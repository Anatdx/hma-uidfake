cmake_minimum_required(VERSION 3.20)
project(hma-uidfake LANGUAGES NONE)

# Usage:
#   cmake -S . -B build -DUG_NDK=/path/to/ndk
#   cmake --build build                     # default target: flashable zip
#   cmake --build build --target ko         # all KMI modules
#   cmake --build build --target ko-android14-6.1
#   cmake --build build --target sync-tool
# Everything is written into build/ only: ko/  tools-build/  dist/

set(DDK_ROOT "/opt/ddk" CACHE PATH "DDK root (provides kdir/ and clang/)")
set(UG_NDK "" CACHE PATH "Android NDK root")
option(UG_USE_PREBUILT_KO "Package existing kernel modules from build/ko" OFF)
option(UG_DEBUG "Build a module whose diagnostics are on from boot" OFF)
set(UG_VERSION_OVERRIDE "" CACHE STRING "Explicit package version, e.g. 0.4.0-dev+inline-local")
set(UG_VERSION_CODE_OVERRIDE "" CACHE STRING "Explicit numeric versionCode for exported source builds")

# KMI -> clang shipped with the DDK. Each KMI is built by the compiler its kernel was built with,
# which is also the compiler clang CFI and the shadow call stack were generated by.
set(UG_KMI_CLANG
    android14-6.1:clang-r487747c
    android15-6.6:clang-r510928
    android16-6.12:clang-r536225
    android17-6.18:clang-r584948c)
# The 5.x kernels are built only when asked for. One of them was reported not to boot,
# which is what the write guards in patch.c are for: a target outside [_stext, _end) or a
# physical address outside the image is refused, so a wrong assumption there now fails to
# hook instead of corrupting a page. Enable this to test a 5.x device.
option(UG_ENABLE_5X "Also build the 5.x kernel modules" OFF)
if(UG_ENABLE_5X)
  list(PREPEND UG_KMI_CLANG
      android12-5.10:clang-r416183b
      android13-5.10:clang-r450784e
      android13-5.15:clang-r450784e
      android14-5.15:clang-r487747c)
endif()

if(NOT UG_NDK)
  set(UG_NDK "$ENV{ANDROID_NDK_HOME}")
endif()
if(NOT UG_NDK)
  message(FATAL_ERROR "Android NDK required: -DUG_NDK=/path/to/ndk or set ANDROID_NDK_HOME")
endif()
if(NOT EXISTS "${UG_NDK}/build/cmake/android.toolchain.cmake")
  message(FATAL_ERROR "Android NDK toolchain not found under ${UG_NDK}")

if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/external/lkmloader/CMakeLists.txt")

# module id / version come from module/module.prop
# The version git reports: <tag without its v> + the short hash, e.g. 0.1.0+a77bea6. module.prop
# carries only the fallback, for a source tree without git (a release tarball).
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
             "${CMAKE_CURRENT_SOURCE_DIR}/module/module.prop")
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/module/module.prop" _prop)
foreach(_l IN LISTS _prop)
  if(_l MATCHES "^id=(.+)$")
    set(UG_ID "${CMAKE_MATCH_1}")
  elseif(_l MATCHES "^version=(.+)$")
    set(UG_BASE "${CMAKE_MATCH_1}")
  elseif(_l MATCHES "^versionCode=(.+)$")
    set(UG_BASE_CODE "${CMAKE_MATCH_1}")
  endif()
endforeach()

function(ug_git out)
  set(${out} "" PARENT_SCOPE)
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/.git")
    return()
  endif()
  execute_process(COMMAND git -C "${CMAKE_CURRENT_SOURCE_DIR}" ${ARGN}
                  OUTPUT_VARIABLE _out OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET
                  RESULT_VARIABLE _rc)
  if(_rc EQUAL 0)
    set(${out} "${_out}" PARENT_SCOPE)
  endif()
endfunction()

ug_git(_tag describe --tags --abbrev=0)
ug_git(_hash rev-parse --short HEAD)
if(_tag)
  string(REGEX REPLACE "^v" "" _tag "${_tag}")
else()
  set(_tag "${UG_BASE}")
endif()
if(_hash)
  ug_git(_dirty status --porcelain)
  if(_dirty)
    set(_hash "${_hash}-dirty")
  endif()
  set(UG_VER "${_tag}+${_hash}")
else()
  set(UG_VER "${_tag}")
endif()
if(NOT UG_VERSION_OVERRIDE STREQUAL "")
  if(NOT UG_VERSION_OVERRIDE MATCHES "^[A-Za-z0-9][A-Za-z0-9._+-]*$")
    message(FATAL_ERROR "UG_VERSION_OVERRIDE may only contain letters, numbers, '.', '_', '+', and '-'")
  endif()
  set(UG_VER "${UG_VERSION_OVERRIDE}")
endif()

# versionCode has to be a number and has to grow: it is the tag as
# major * 1000000 + minor * 1000 + patch, so 0.1.0 is 1000 and a new tag is the only thing that
# raises it. A tag that is not x.y.z falls back to the value module.prop carries.
if(_tag MATCHES "^([0-9]+)\\.([0-9]+)\\.([0-9]+)$")
  math(EXPR UG_VERCODE "${CMAKE_MATCH_1} * 1000000 + ${CMAKE_MATCH_2} * 1000 + ${CMAKE_MATCH_3}")
else()
  set(UG_VERCODE "${UG_BASE_CODE}")
endif()
if(NOT UG_VERSION_CODE_OVERRIDE STREQUAL "")
  if(NOT UG_VERSION_CODE_OVERRIDE MATCHES "^(0|[1-9][0-9]*)$")
    message(FATAL_ERROR "UG_VERSION_CODE_OVERRIDE must be a nonnegative decimal integer")
  endif()
  set(UG_VERCODE "${UG_VERSION_CODE_OVERRIDE}")
endif()

# A commit on the checked-out branch has to re-run this, or the version would be stale.
# Git worktrees have a .git file; exported source archives may have no .git at all.
# Let git resolve both the worktree HEAD and the shared refs directory.
foreach(_ref HEAD packed-refs)
  ug_git(_ref_path rev-parse --git-path "${_ref}")
  if(_ref_path)
    get_filename_component(_ref_path "${_ref_path}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
    if(EXISTS "${_ref_path}")
      set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_ref_path}")
    endif()
  endif()
endforeach()
ug_git(_head_ref symbolic-ref -q HEAD)
if(_head_ref)
  ug_git(_head_path rev-parse --git-path "${_head_ref}")
  get_filename_component(_head_path "${_head_path}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
  if(EXISTS "${_head_path}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_head_path}")
  endif()
endif()

# The module.prop that gets installed carries the version the module manager shows.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/module/module.prop" _prop_text)
# By value, not by line anchor: file(READ) hands over the whole file and CMake's ^ matches only
# the start of that string.
string(REPLACE "version=${UG_BASE}" "version=${UG_VER}" _prop_text "${_prop_text}")
string(REPLACE "versionCode=${UG_BASE_CODE}" "versionCode=${UG_VERCODE}" _prop_text "${_prop_text}")
file(WRITE "${CMAKE_BINARY_DIR}/module.prop" "${_prop_text}")
set(KO_NAME hma_uidfake)

set(KO_DIR "${CMAKE_BINARY_DIR}/ko")
file(MAKE_DIRECTORY "${KO_DIR}")
file(GLOB SRC_FILES CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/src/*.c"
    "${CMAKE_CURRENT_SOURCE_DIR}/src/*.S"
    "${CMAKE_CURRENT_SOURCE_DIR}/src/include/*.h")
list(APPEND SRC_FILES "${CMAKE_CURRENT_SOURCE_DIR}/src/Makefile")
list(APPEND SRC_FILES "${CMAKE_CURRENT_SOURCE_DIR}/scripts/build-kmi.sh")

# Kernel modules must be linked by kbuild (modpost, vermagic, symbol CRCs). kbuild
# writes its objects into the M= directory, so every KMI builds inside
# build/kmi/<kmi>/ -- a directory that only holds symlinks to src/. The source tree
# is therefore never written to, and no copy of it is needed either.
set(_ko_outputs "")
if(UG_USE_PREBUILT_KO)
  file(GLOB _ko_outputs CONFIGURE_DEPENDS "${KO_DIR}/*.ko")
  if(NOT _ko_outputs)
    message(FATAL_ERROR "UG_USE_PREBUILT_KO is enabled, but no kernel modules were found in ${KO_DIR}")
  endif()
else()
  # CMAKE_MAKE_PROGRAM can be Ninja. Kbuild always requires GNU make.
  find_program(UG_KBUILD_MAKE NAMES gmake make REQUIRED)
  find_program(UG_BASH NAMES bash REQUIRED)
  foreach(_pair IN LISTS UG_KMI_CLANG)
    string(REPLACE ":" ";" _p "${_pair}")
    list(GET _p 0 _kmi)
    list(GET _p 1 _clang)
    set(_kdir "${DDK_ROOT}/kdir/${_kmi}")
    set(_clang_bin "${DDK_ROOT}/clang/${_clang}/bin")
    set(_stage "${CMAKE_BINARY_DIR}/kmi/${_kmi}")
    set(_out "${KO_DIR}/${_kmi}_arm64_${KO_NAME}.ko")

    if(NOT EXISTS "${_kdir}" OR NOT EXISTS "${_clang_bin}")
      message(FATAL_ERROR "missing DDK tree or clang for ${_kmi}: ${_kdir} / ${_clang_bin}")
    endif()

    add_custom_command(OUTPUT "${_out}"
      COMMAND ${CMAKE_COMMAND} -E env "DDK_ROOT=${DDK_ROOT}" "UG_KBUILD_MAKE=${UG_KBUILD_MAKE}"
              "${UG_BASH}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/build-kmi.sh"
              "${_kmi}" "${_clang}" "${_stage}" "${_out}" "$<BOOL:${UG_DEBUG}>"
      DEPENDS ${SRC_FILES}
      COMMENT "kbuild ${_kmi}"
      VERBATIM)
    add_custom_target("ko-${_kmi}" DEPENDS "${_out}")
    list(APPEND _ko_outputs "${_out}")
  endforeach()
endif()
add_custom_target(ko DEPENDS ${_ko_outputs})

# sync-tool: src/tools is a standalone CMake project, cross compiled with the NDK.
file(GLOB TOOL_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/tools/*")
file(GLOB TOOL_HEADERS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/include/*.h")
set(UG_TOOLS_BUILD "${CMAKE_BINARY_DIR}/tools-build")
set(UG_SYNC "${UG_TOOLS_BUILD}/sync-tool")
# The NDK toolchain restricts find_package to the sysroot by default; nlohmann_json
# comes from the host distro, so widen the package search scope.
add_custom_command(OUTPUT "${UG_SYNC}"
  COMMAND ${CMAKE_COMMAND} -S "${CMAKE_CURRENT_SOURCE_DIR}/src/tools" -B "${UG_TOOLS_BUILD}"
          -DCMAKE_TOOLCHAIN_FILE=${UG_NDK}/build/cmake/android.toolchain.cmake
          -DANDROID_ABI=arm64-v8a -DANDROID_PLATFORM=24 -DCMAKE_BUILD_TYPE=Release
          -DCMAKE_FIND_ROOT_PATH_MODE_PACKAGE=BOTH
  COMMAND ${CMAKE_COMMAND} --build "${UG_TOOLS_BUILD}"
  DEPENDS ${TOOL_SOURCES} ${TOOL_HEADERS}
  COMMENT "building sync-tool (CMake + NDK)"
  VERBATIM)
add_custom_target(sync-tool DEPENDS "${UG_SYNC}")

# lkmloader: external/lkmloader is a git submodule (see .gitmodules, and the gitlink this
# commit pins it to) and a standalone CMake project. It loads the ko by
# resolving undefined symbols from /proc/kallsyms and calling init_module, and rewrites the
# vermagic once when the kernel names the value it wants -- the same two things ksud insmod
# does, with no dependency on KernelSU being installed.
set(UG_LKM_BUILD "${CMAKE_BINARY_DIR}/lkmloader-build")
set(UG_LKM "${UG_LKM_BUILD}/lkmloader")
file(GLOB_RECURSE LKM_SOURCES CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/external/lkmloader/*.c"
    "${CMAKE_CURRENT_SOURCE_DIR}/external/lkmloader/*.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/external/lkmloader/*.h"
    "${CMAKE_CURRENT_SOURCE_DIR}/external/lkmloader/*.hpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/external/lkmloader/*.cmake"
    "${CMAKE_CURRENT_SOURCE_DIR}/external/lkmloader/CMakeLists.txt")

# Packaging: module/ template + ko + sync-tool into one zip (CMake's own tar).
set(UG_STAGE "${CMAKE_BINARY_DIR}/stage")
# Only top-level template files belong in the normal package. Experimental
# templates may live in subdirectories and must not become release payloads.
file(GLOB UG_MODULE_FILES CONFIGURE_DEPENDS LIST_DIRECTORIES false
     "${CMAKE_CURRENT_SOURCE_DIR}/module/*")
set(UG_MODULE_ITEMS)
foreach(_file IN LISTS UG_MODULE_FILES)
  get_filename_component(_name "${_file}" NAME)
  list(APPEND UG_MODULE_ITEMS "${_name}")
endforeach()
set(UG_ZIP "${CMAKE_BINARY_DIR}/dist/${UG_ID}-${UG_VER}.zip")

add_custom_command(OUTPUT "${UG_STAGE}/.stamp"
  COMMAND ${CMAKE_COMMAND} -E rm -rf "${UG_STAGE}"
  COMMAND ${CMAKE_COMMAND} -E make_directory "${UG_STAGE}/ko"
  COMMAND ${CMAKE_COMMAND} -E copy ${UG_MODULE_FILES} "${UG_STAGE}"
  COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_BINARY_DIR}/module.prop" "${UG_STAGE}/module.prop"
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${KO_DIR}" "${UG_STAGE}/ko"
  COMMAND ${CMAKE_COMMAND} -E copy "${UG_SYNC}" "${UG_STAGE}/sync-tool"
  COMMAND ${CMAKE_COMMAND} -E copy "${UG_LKM}" "${UG_STAGE}/lkmloader"
  COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/external/lkmloader/LICENSE" "${UG_STAGE}/lkmloader.LICENSE"
  COMMAND ${CMAKE_COMMAND} -E touch "${UG_STAGE}/.stamp"
  DEPENDS ${_ko_outputs} "${UG_SYNC}" "${UG_LKM}" "${CMAKE_BINARY_DIR}/module.prop"
          "${CMAKE_CURRENT_SOURCE_DIR}/external/lkmloader/LICENSE" ${UG_MODULE_FILES}
  COMMENT "staging module contents"
  VERBATIM)

add_custom_command(OUTPUT "${UG_ZIP}"
  COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/dist"
  COMMAND ${CMAKE_COMMAND} -E rm -f "${UG_ZIP}"
  COMMAND ${CMAKE_COMMAND} -E tar cf "${UG_ZIP}" --format=zip -- ${UG_MODULE_ITEMS} sync-tool lkmloader lkmloader.LICENSE ko
  WORKING_DIRECTORY "${UG_STAGE}"
  DEPENDS "${UG_STAGE}/.stamp"
  COMMENT "packing ${UG_ZIP}"
  VERBATIM)

add_custom_command(OUTPUT "${UG_LKM}"
  COMMAND ${CMAKE_COMMAND} -S "${CMAKE_CURRENT_SOURCE_DIR}/external/lkmloader" -B "${UG_LKM_BUILD}"
          -DCMAKE_TOOLCHAIN_FILE=${UG_NDK}/build/cmake/android.toolchain.cmake
          -DANDROID_ABI=arm64-v8a -DANDROID_PLATFORM=24 -DCMAKE_BUILD_TYPE=Release
  COMMAND ${CMAKE_COMMAND} --build "${UG_LKM_BUILD}"
  DEPENDS ${LKM_SOURCES}
  COMMENT "building lkmloader (CMake + NDK)"
  VERBATIM)
add_custom_target(lkmloader DEPENDS "${UG_LKM}")

add_custom_target(module ALL DEPENDS "${UG_ZIP}")
